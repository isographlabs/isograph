# resolve_position: unmatched span `ResolvedNode` via `From`

`#[resolve_position]` gains `on_unmatched_span = from_path` and `on_unmatched_span = from_parent`. When the wrapper span contained the position and no field span does, `resolve` returns `self.path(parent).to()` or `parent.to()`. The `From` impl at the monomorph chooses the `ResolvedNode` variant. generic-slot.md uses this.

```rust
// from crates/resolve_position_macros/tests/on_unmatched_span.rs
#[derive(Debug, ResolvePosition)]
#[resolve_position(
    parent_type = ParentPath<'a>,
    resolved_node = TestResolvedNode<'a>,
    on_unmatched_span = from_path
)]
struct Slot<T: ResolvePosition>
where
    for<'a> T: ResolvePosition<ResolvedNode<'a> = TestResolvedNode<'a>>,
    for<'a> TestResolvedNode<'a>:
        From<PositionResolutionPath<&'a Slot<T>, <T as ResolvePosition>::Parent<'a>>>,
{
    #[resolve_field]
    item: Option<WithSpan<T>>,
}
```

```rust
// generated by resolve_position_macros/src/resolve_position_macro.rs
impl<T: ResolvePosition> ::resolve_position::ResolvePosition for Slot<T>
where
    for<'a> T: ResolvePosition<ResolvedNode<'a> = TestResolvedNode<'a>>,
    for<'a> TestResolvedNode<'a>:
        From<PositionResolutionPath<&'a Slot<T>, <T as ResolvePosition>::Parent<'a>>>,
{
    type Parent<'a>
        = ParentPath<'a>
    where
        Self: 'a;
    type ResolvedNode<'a>
        = TestResolvedNode<'a>
    where
        Self: 'a;

    fn resolve<'a>(
        &'a self,
        parent: Self::Parent<'a>,
        position: ::span::Span,
    ) -> Self::ResolvedNode<'a> {
        for item in self.item.iter() {
            if item.location.contains(position) {
                let new_parent = self.path(parent);
                return item.item.resolve(new_parent, position);
            }
        }
        return self.path(parent).to();
    }
}
```

```rust
// from crates/resolve_position_macros/tests/on_unmatched_span.rs
impl<'a> From<PositionResolutionPath<&'a Slot<Child>, ParentPath<'a>>> for TestResolvedNode<'a> {
    fn from(path: PositionResolutionPath<&'a Slot<Child>, ParentPath<'a>>) -> Self {
        TestResolvedNode::Slot(path)
    }
}
```

`on_unmatched_span = from_parent` is the same arm with `return parent.to();` and `ResolvedNode: From<Parent>`.

Absent `on_unmatched_span`, the unmatched-span arm is unchanged:

```rust
// generated by resolve_position_macros/src/resolve_position_macro.rs
            return Self::ResolvedNode::EntrypointDeclaration(self.path(parent).to());
```

`transparent` still emits no unmatched-span arm. `on_unmatched_span` on an enum is a compile error. Any ident other than `from_path` or `from_parent` is a compile error.

## Tests

```rust
// from crates/resolve_position_macros/tests/on_unmatched_span.rs
use prelude::Postfix;
use resolve_position::{PositionResolutionPath, ResolvePosition};
use resolve_position_macros::ResolvePosition;
use span::{Span, WithSpan, WithSpanPostfix};

#[derive(Debug)]
enum TestResolvedNode<'a> {
    List(ParentPath<'a>),
    Slot(PositionResolutionPath<&'a Slot<Child>, ParentPath<'a>>),
    Child(ChildPath<'a>),
}

#[derive(Debug, ResolvePosition)]
#[resolve_position(
    parent_type = ParentPath<'a>,
    resolved_node = TestResolvedNode<'a>,
    on_unmatched_span = from_path
)]
struct Slot<T: ResolvePosition>
where
    for<'a> T: ResolvePosition<ResolvedNode<'a> = TestResolvedNode<'a>>,
    for<'a> TestResolvedNode<'a>:
        From<PositionResolutionPath<&'a Slot<T>, <T as ResolvePosition>::Parent<'a>>>,
{
    #[resolve_field]
    item: Option<WithSpan<T>>,
}

#[derive(Debug, ResolvePosition)]
#[resolve_position(
    parent_type = ParentPath<'a>,
    resolved_node = TestResolvedNode<'a>,
    on_unmatched_span = from_parent
)]
struct Wrapper {
    #[resolve_field]
    item: Option<WithSpan<Child>>,
}

#[derive(Debug, ResolvePosition)]
#[resolve_position(parent_type = (), resolved_node = TestResolvedNode<'a>)]
struct List(#[resolve_field] Vec<WithSpan<Slot<Child>>>);

#[derive(Debug, ResolvePosition)]
#[resolve_position(parent_type = ParentPath<'a>, resolved_node = TestResolvedNode<'a>)]
struct Child;

type ParentPath<'a> = PositionResolutionPath<&'a List, ()>;

type ChildPath<'a> = PositionResolutionPath<&'a Child, ParentPath<'a>>;

impl<'a> From<PositionResolutionPath<&'a Slot<Child>, ParentPath<'a>>> for TestResolvedNode<'a> {
    fn from(path: PositionResolutionPath<&'a Slot<Child>, ParentPath<'a>>) -> Self {
        TestResolvedNode::Slot(path)
    }
}

impl<'a> From<ParentPath<'a>> for TestResolvedNode<'a> {
    fn from(path: ParentPath<'a>) -> Self {
        TestResolvedNode::List(path)
    }
}

#[test]
fn from_path_unmatched_span_uses_the_from_impl() {
    let list = List(
        Slot::<Child> {
            item: Child.with_span(Span::new(0, 2)).wrap_some(),
        }
        .with_span(Span::new(0, 8))
        .wrap_vec(),
    );

    match list.resolve((), Span::new(4, 5)) {
        TestResolvedNode::Slot(path) => {
            assert!(path.inner.item.is_some());
        }
        node => panic!("expected the slot, got {node:?}"),
    }
}

#[test]
fn from_path_matched_field_descends_into_the_child() {
    let list = List(
        Slot::<Child> {
            item: Child.with_span(Span::new(0, 4)).wrap_some(),
        }
        .with_span(Span::new(0, 4))
        .wrap_vec(),
    );

    match list.resolve((), Span::new(1, 2)) {
        TestResolvedNode::Child(_) => {}
        node => panic!("expected the child, got {node:?}"),
    }
}

#[test]
fn from_parent_unmatched_span_uses_the_from_impl() {
    let list = List(vec![]);
    let wrapper = Wrapper {
        item: Child.with_span(Span::new(0, 2)).wrap_some(),
    };

    match wrapper.resolve(list.path(()), Span::new(4, 5)) {
        TestResolvedNode::List(path) => {
            assert!(std::ptr::eq(path.inner, list.reference()));
        }
        node => panic!("expected the list, got {node:?}"),
    }
}
```
